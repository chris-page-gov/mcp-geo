import json, sys

def extract_segs(json_text):
    """Extract LineString coordinate arrays from GeoJSON features text"""
    try:
        data = json.loads(json_text)
    except json.JSONDecodeError as e:
        print(f"JSON decode error: {e}", file=sys.stderr)
        return []
    segs = []
    for feat in data.get('features', []):
        geom = feat.get('geometry', {})
        coords = geom.get('coordinates', [])
        if geom.get('type') == 'LineString' and len(coords) >= 2:
            segs.append(coords)
    return segs

def segs_to_js(segs):
    """Convert coordinate arrays to compact JS array literal"""
    parts = []
    for seg in segs:
        pts = ','.join(f'[{c[0]},{c[1]}]' for c in seg)
        parts.append(f'[{pts}]')
    return '[' + ',\n    '.join(parts) + ']'

        all_segs.extend(segs)
        print(f"  {f}: {len(segs)} segs", file=sys.stderr)
